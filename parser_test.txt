// 正确用例
// 1.while语句
{
    int i;int j;float v;float x;float[100] a;
    while(true){
        do i = i + 1;while(a[i]<v);
        do j = j + 1;while(a[j]>v);
        if(i>=j)break;
        x=a[i];a[i]=a[j];a[j]=x;
    }
}

// 2.新增for语句
{
	int i;int j;float v;float x;float[100] a;
	for(;true;){
		for(;a[i]<v;)i = i + 1;
		for(;a[j]>v;)j = j + 1;
		if(i>=j)break;
		x=a[i];a[i]=a[j];a[j]=x;
	}
}

// 错误用例
{
	    int i;int j;float v;float x;float[100] a;
	    for(;true;){
	    	for(;a[i]<v;)i == i + 1;  // ==语法错误
	    	for(;a[j]>v;)j = j + 1;
	    	if(i>=j)break;
	    	x=a[i];a[i]=a[j];a[j]=x;
	    }
}
